# Setup Wizard Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a full-screen 4-step setup wizard (SCION folder → topology → config → loading) that runs before the main AppShell, using TanStack Router for routing and Tauri plugin-store for recent-item persistence.

**Architecture:** Two TanStack Router routes — `/setup` (wizard, no AppShell) and `/app` (AppShell with child routes). The wizard runs `create_topology_config` (streams `scion.sh` output live), copies the generated `gen/` folder to app data dir, then calls `load_project` on the copy. The resulting topology is stored in a module-level session object and read by the `/app/topology` route. `App.tsx` is deleted; its AppShell moves to `src/routes/app.tsx`.

**Tech Stack:** React 19, TypeScript, Mantine 9, TanStack Router v1, `@tauri-apps/plugin-store`, Tauri 2, tokio (process + io-util features)

```rust
#[tauri::command]
#[specta::specta]
fn load_project(folder: String, topo_file: String) -> Result<Topology, String> {
    let output = std::process::Command::new("./scion.sh")
        .args(["topology", "-c", &topo_file])
        .current_dir(&folder)
        .output()
        .map_err(|e| format!("Failed to run scion.sh: {e}"))?;

    if !output.status.success() {
        return Err(format!(
            "scion.sh topology failed:\n{}",
            String::from_utf8_lossy(&output.stderr)
        ));
    }

    let pattern = format!("{folder}/gen/AS*/topology.json");
    let raw_ases: Vec<RawAsTopology> = glob::glob(&pattern)
        .map_err(|e| format!("Invalid glob pattern: {e}"))?
        .map(|entry| {
            let path = entry.map_err(|e| e.to_string())?;
            let content = std::fs::read_to_string(&path)
                .map_err(|e| format!("Failed to read {}: {e}", path.display()))?;
            serde_json::from_str::<RawAsTopology>(&content)
                .map_err(|e| format!("Failed to parse {}: {e}", path.display()))
        })
        .collect::<Result<Vec<_>, String>>()?;

    Ok(build_topology(raw_ases))
}

#[tauri::command]
#[specta::specta]
async fn create_topology_config(
    app: tauri::AppHandle,
    folder: String,
    topo_file: String,
) -> Result<String, String> {
    use std::process::Stdio;
    use tokio::io::{AsyncBufReadExt, BufReader};

    let mut child = tokio::process::Command::new("./scion.sh")
        .args(["topology", "-c", &topo_file])
        .current_dir(&folder)
        .stdout(Stdio::piped())
        .stderr(Stdio::piped())
        .spawn()
        .map_err(|e| format!("Failed to spawn scion.sh: {e}"))?;

    let stdout = child.stdout.take().unwrap();
    let stderr = child.stderr.take().unwrap();

    let app_out = app.clone();
    let stdout_task = tokio::spawn(async move {
        let mut lines = BufReader::new(stdout).lines();
        while let Ok(Some(line)) = lines.next_line().await {
            app_out.emit("topology-config-output", line).ok();
        }
    });

    let app_err = app.clone();
    let stderr_task = tokio::spawn(async move {
        let mut lines = BufReader::new(stderr).lines();
        while let Ok(Some(line)) = lines.next_line().await {
            app_err.emit("topology-config-output", line).ok();
        }
    });

    let status = child
        .wait()
        .await
        .map_err(|e| format!("Failed to wait for scion.sh: {e}"))?;
    let _ = tokio::join!(stdout_task, stderr_task);

    if !status.success() {
        return Err(format!("scion.sh exited with status {status}"));
    }

    let src_gen = std::path::Path::new(&folder).join("gen");
    let app_data_dir = app
        .path()
        .app_data_dir()
        .map_err(|e| format!("Cannot resolve app data dir: {e}"))?;
    let project_name = std::path::Path::new(&folder)
        .file_name()
        .and_then(|n| n.to_str())
        .unwrap_or("project");
    let dst_gen = app_data_dir
        .join("projects")
        .join(project_name)
        .join("gen");

    copy_dir_all(&src_gen, &dst_gen)?;

    dst_gen
        .to_str()
        .map(|s| s.to_string())
        .ok_or_else(|| "Generated path contains invalid Unicode".to_string())
}

fn copy_dir_all(src: &std::path::Path, dst: &std::path::Path) -> Result<(), String> {
    std::fs::create_dir_all(dst)
        .map_err(|e| format!("Cannot create {}: {e}", dst.display()))?;
    for entry in std::fs::read_dir(src)
        .map_err(|e| format!("Cannot read {}: {e}", src.display()))?
    {
        let entry = entry.map_err(|e| e.to_string())?;
        let ty = entry.file_type().map_err(|e| e.to_string())?;
        if ty.is_dir() {
            copy_dir_all(&entry.path(), &dst.join(entry.file_name()))?;
        } else {
            std::fs::copy(entry.path(), dst.join(entry.file_name()))
                .map_err(|e| e.to_string())?;
        }
    }
    Ok(())
}

#[tauri::command]
#[specta::specta]
fn list_topology_files(folder: String) -> Result<Vec<String>, String> {
    let topo_dir = std::path::Path::new(&folder).join("topology");
    if !topo_dir.exists() {
        return Ok(Vec::new());
    }
    let mut files = Vec::new();
    for entry in std::fs::read_dir(&topo_dir)
        .map_err(|e| format!("Cannot read topology dir: {e}"))?
    {
        let entry = entry.map_err(|e| e.to_string())?;
        let path = entry.path();
        if path.extension().and_then(|e| e.to_str()) == Some("topo") {
            if let Some(s) = path.to_str() {
                files.push(s.to_string());
            }
        }
    }
    Ok(files)
}
```

```tsx
  beforeLoad: () => {
    throw redirect({ to: "/setup" });

const setupRoute = createRoute({
  getParentRoute: () => rootRoute,

  // Redirect to app if a topology is already loaded in this session

    if (session.getTopology()) {
      throw redirect({ to: "/app/topology" });
    }
  },
  component: SetupWizard,
});

const appRoute = createRoute({
  getParentRoute: () => rootRoute,
  path: "/app",
  component: AppLayout,
});

const appIndexRoute = createRoute({
  getParentRoute: () => appRoute,
  path: "/",
  beforeLoad: () => {
    throw redirect({ to: "/app/topology" });
  },
});

const routeTree = rootRoute.addChildren([
  indexRoute,
  setupRoute,
  appRoute.addChildren([appIndexRoute]),
]);
```

```tsx
export function TopologyView({ topology }: Props) {
  const [nodes, setNodes, onNodesChange] = useNodesState<TopologyNode>([]);
  const [edges, setEdges, onEdgesChange] = useEdgesState<TopologyEdge>([]);
  const [selected, setSelected] = useState<Selected>(null);

  useEffect(() => {
    const { nodes: graphNodes, edges: graphEdges } = buildGraph(topology);
    const laidOut = layoutGraph(graphNodes, graphEdges);
    setNodes(laidOut);
    setEdges(graphEdges);
  }, [topology, setNodes, setEdges]);

  const onNodeClick: NodeMouseHandler<TopologyNode> = useCallback(
    (_: React.MouseEvent, node: TopologyNode) => setSelected(node),
    []
  );
  const onEdgeClick: EdgeMouseHandler<TopologyEdge> = useCallback(
    (_: React.MouseEvent, edge: TopologyEdge) => setSelected(edge),
    []
  );

  const styledEdges = useMemo(
    () =>
      edges.map((edge) => ({
        ...edge,
        ...(edgeStyle[edge.data?.linkType ?? "Core"] ?? {}),

    [edges]
  );

  return (
    <div style={{ width: "100%", height: "100%", position: "relative" }}>
      <ReactFlow
        nodes={nodes}
        edges={styledEdges}
        nodeTypes={nodeTypes}
        onNodesChange={onNodesChange}
        onEdgesChange={onEdgesChange}
        onNodeClick={onNodeClick}
        onEdgeClick={onEdgeClick}
        onPaneClick={() => setSelected(null)}
        fitView

        <Controls />

        title="Details"
```

```ts
import { Store } from "@tauri-apps/plugin-store";

const STORE_FILE = "recent.json";
const MAX_RECENT = 10;

type StoreKey = "recent-folders" | "recent-topologies";

async function getStore() {
  return Store.load(STORE_FILE);
}

export async function getRecent(key: StoreKey): Promise<string[]> {
  const store = await getStore();
  return (await store.get<string[]>(key)) ?? [];
}

export async function pushRecent(key: StoreKey, value: string): Promise<void> {
  const store = await getStore();
  const existing = (await store.get<string[]>(key)) ?? [];
  const deduped = [value, ...existing.filter((v) => v !== value)].slice(
    0,
    MAX_RECENT
  );
  await store.set(key, deduped);
  await store.save();
}
```

```tsx
function LoadingStep({
  folder,
  topoFile,
  onBack,
}: {
  folder: string;
  topoFile: string;
  onBack: () => void;
}) {
  const navigate = useNavigate();
  const [lines, setLines] = useState<string[]>([]);
  const [error, setError] = useState<string | null>(null);
  const viewportRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    let unlisten: (() => void) | undefined;
    let cancelled = false;

    async function run() {
      unlisten = await listen<string>("topology-config-output", (e) => {
        if (cancelled) return;
        setLines((prev) => {
          const next = [...prev, e.payload];
          // scroll after state update
          setTimeout(() => {
            if (viewportRef.current) {
              viewportRef.current.scrollTop =
                viewportRef.current.scrollHeight;
            }
          }, 0);
          return next;
        });
      });

      const configResult = await commands.createTopologyConfig(folder, topoFile);
      if (cancelled) return;
      if (configResult.status === "error") {
        setError(configResult.error);
        return;
      }

      const topoResult = await commands.loadProject(configResult.data);
      if (cancelled) return;
      if (topoResult.status === "error") {
        setError(topoResult.error);
        return;
      }

      await pushRecent("recent-folders", folder);
      await pushRecent("recent-topologies", topoFile);
      session.setTopology(topoResult.data);
      navigate({ to: "/app/topology" });
    }

    run();

    return () => {
      cancelled = true;
      unlisten?.();
    };
  }, [folder, topoFile, navigate]);

  if (error) {
    return (
      <Stack gap="md" mt="md">
        <Text c="red" size="sm" style={{ wordBreak: "break-word" }}>
          {error}
        </Text>
        <Button variant="default" onClick={onBack}>
          Start Over
        </Button>
      </Stack>
    );
  }

  return (
    <Stack gap="md" mt="md">
      <ScrollArea h={300} viewportRef={viewportRef}>
        <Code
          block
          style={{
            fontFamily: "monospace",
            whiteSpace: "pre-wrap",
            wordBreak: "break-all",
            minHeight: 300,
          }}
        >
          {lines.length > 0 ? lines.join("\n") : "Running…"}
        </Code>
      </ScrollArea>
    </Stack>
  );
}
```
